# Test Fixtures

Fixtures are static assets (images, audio clips, JSON payloads) loaded at runtime by scenario tests to ensure **repeatable, offline-friendly** execution.

:::tip
**Benefits:**

- **Determinism**: The exact same bytes are fed to the agent and judge on every run.
- **Speed & Cost**: No network fetches or external dependencies.
- **Debugging**: Easy to reproduce failures locally.
  :::

## Fixture Directory Structure

- Create a `fixtures/` directory **next to your test file**.
- Commit the asset (e.g. `scenario.webp`) to version control so CI has it.
- Keep binary files small—**&lt;100&nbsp;KB for images** is a good rule of thumb.

```text
javascript/examples/vitest/tests/
└─ fixtures/
   └─ scenario.webp
```

```typescript [typescript]
import * as fs from "fs";
import * as path from "path";
import { describe, it, expect } from "vitest";
import scenario, { type AgentAdapter, AgentRole } from "@langwatch/scenario";

function getDataURLFromFixture(filename: string, filetype: string): string {
  const filePath = path.join(__dirname, "fixtures", filename);
  const buffer = fs.readFileSync(filePath);
  return `data:${filetype};base64,${buffer.toString("base64")}`;
}

// Example agent that can analyze images
const imageAnalysisAgent: AgentAdapter = {
  role: AgentRole.AGENT,
  call: async (input) => {
    // This would be your actual agent implementation
    // For demo purposes, we'll return a simple response
    return "I can see an image in the conversation. This appears to be a screenshot or diagram.";
  },
};

describe("Image Analysis with Fixtures", () => {
  it("should analyze an image from fixtures", async () => {
    const result = await scenario.run({
      name: "image analysis test",
      description: "Test the agent's ability to analyze images from fixtures",
      agents: [
        imageAnalysisAgent,
        scenario.userSimulatorAgent(),
        scenario.judgeAgent(),
      ],
      script: [
        {
          role: "user",
          content: [
            { type: "text", text: "What do you see in this image?" },
            {
              type: "image",
              image: getDataURLFromFixture("scenario.webp", "image/webp"),
            },
          ],
        },
        scenario.agent(),
        scenario.judge({
          criteria: [
            "Agent should acknowledge the presence of an image",
            "Agent should provide some analysis or description",
          ],
        }),
      ],
    });

    expect(result.success).toBe(true);
  });
});
```

```go [go]
package examples

import (
	"context"
	"os"
	"path/filepath"
	"testing"

	scenario "github.com/langwatch/scenario/go"
)

// go test runs in the package directory, so fixtures/ resolves next to the test file
func readFixture(t *testing.T, filename string) []byte {
	t.Helper()
	data, err := os.ReadFile(filepath.Join("fixtures", filename))
	if err != nil {
		t.Fatal(err)
	}
	return data
}

// Example agent that can analyze images
var imageAnalysisAgent = scenario.AgentFunc(func(ctx context.Context, input scenario.AgentInput) (*scenario.AgentReturn, error) {
	// This would be your actual agent implementation
	// For demo purposes, we'll return a simple response
	return scenario.TextReturn("I can see an image in the conversation. This appears to be a screenshot or diagram."), nil
})

func TestImageAnalysisWithFixtures(t *testing.T) {
	result, err := scenario.Run(context.Background(), scenario.ScenarioConfig{
		Name:        "image analysis test",
		Description: "Test the agent's ability to analyze images from fixtures",
		Agents: []scenario.AgentAdapter{
			imageAnalysisAgent,
			scenario.NewUserSimulatorAgent(scenario.UserSimulatorAgentConfig{}),
			scenario.NewJudgeAgent(scenario.JudgeAgentConfig{}),
		},
		Script: []scenario.ScriptStep{
			scenario.MessageStep(scenario.Message{
				Role: scenario.RoleUser,
				Parts: []scenario.Part{
					scenario.TextPart("What do you see in this image?"),
					scenario.ImagePart("image/webp", readFixture(t, "scenario.webp")),
				},
			}),
			scenario.Agent(),
			scenario.Judge(scenario.WithCriteria(
				"Agent should acknowledge the presence of an image",
				"Agent should provide some analysis or description",
			)),
		},
	})
	if err != nil {
		t.Fatal(err)
	}
	if !result.Success {
		t.Fatalf("scenario failed: %s", result.Reasoning)
	}
}
```
