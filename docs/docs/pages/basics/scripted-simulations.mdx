# Scripted Simulations: Precise Flow Control

Automatic simulations provide powerful testing capabilities. For scenarios requiring precise control, use scripted simulations to orchestrate exactly how conversations unfold, when evaluations occur, and what custom logic runs at each step.

## Script Basics

A script is a list of functions that control the conversation flow:

:::code-group

```python [python]
result = await scenario.run(
    name="scripted interaction",
    description="Test specific conversation patterns",
    agents=[
        MyAgent(),
        scenario.UserSimulatorAgent(),
        scenario.JudgeAgent(),
    ],
    script=[
        scenario.user("Hello, I need help"),
        scenario.agent(),
        scenario.judge(criteria=["Agent responds helpfully"]),
    ]
)

assert result.success
```

```typescript [typescript]
const result = await scenario.run({
  name: "scripted interaction",
  description: "Test specific conversation patterns",
  agents: [
    myAgent,
    scenario.userSimulatorAgent(),
    scenario.judgeAgent(),
  ],
  script: [
    scenario.user("Hello, I need help"),
    scenario.agent(),
    scenario.judge({ criteria: ["Agent responds helpfully"] }),
  ],
});

expect(result.success).toBe(true);
```

```go [go]
result, err := scenario.Run(context.Background(), scenario.ScenarioConfig{
	Name:        "scripted interaction",
	Description: "Test specific conversation patterns",
	Agents: []scenario.AgentAdapter{
		myAgent,
		scenario.NewUserSimulatorAgent(scenario.UserSimulatorAgentConfig{}),
		scenario.NewJudgeAgent(scenario.JudgeAgentConfig{}),
	},
	Script: []scenario.ScriptStep{
		scenario.User("Hello, I need help"),
		scenario.Agent(),
		scenario.Judge(scenario.WithCriteria("Agent responds helpfully")),
	},
})
if err != nil {
	t.Fatal(err)
}
if !result.Success {
	t.Fatalf("scenario failed: %s", result.Reasoning)
}
```

### Commands List

| Command                                                   | Description                                                             |
| --------------------------------------------------------- | ----------------------------------------------------------------------- |
| [`scenario.user()`](#scenariouser)                        | Generate or add user messages                                           |
| [`scenario.agent()`](#scenarioagent)                      | Generate or add agent messages                                          |
| [`scenario.message()`](#scenariomessage)                  | Add any openai-message format messages to the conversation              |
| [`scenario.judge()`](#scenariojudge)                      | Force judge evaluation at specific points                               |
| [`scenario.proceed()`](#scenarioproceed)                  | Let the conversation flow automatically for a specified number of turns |
| [`scenario.succeed()`](#scenariosucceed-and-scenariofail) | End the scenario with a specific result                                 |
| [`scenario.fail()`](#scenariosucceed-and-scenariofail)    | End the scenario with a specific result                                 |

In Go, the commands are `scenario.User()`, `scenario.Agent()`, `scenario.MessageStep()`, `scenario.Judge()`, `scenario.Proceed()`, `scenario.Succeed()` and `scenario.Fail()`.

### <RefLink link={{ python: "script.html#scenario.script.judge", typescript: "functions/judge.html" }}>`scenario.judge()`</RefLink>

Evaluate criteria at specific points in the conversation. When using scripts, pass criteria inline to `scenario.judge()` — this is the preferred approach:

:::code-group

```python [python]
script=[
    scenario.user("I need help"),
    scenario.agent(),

    # Checkpoint: if criteria pass, continue; if any fail, stop immediately
    scenario.judge(criteria=[
        "Agent should ask clarifying questions",
    ]),

    scenario.user(),
    scenario.agent(),

    # Final evaluation with remaining criteria
    scenario.judge(criteria=[
        "Agent provides a helpful response",
        "Agent does not make things up",
    ]),
]
```

```typescript [typescript]
const script = [
  scenario.user("I need help"),
  scenario.agent(),

  // Checkpoint: if criteria pass, continue; if any fail, stop immediately
  scenario.judge({
    criteria: ["Agent should ask clarifying questions"],
  }),

  scenario.user(),
  scenario.agent(),

  // Final evaluation with remaining criteria
  scenario.judge({
    criteria: [
      "Agent provides a helpful response",
      "Agent does not make things up",
    ],
  }),
];
```

```go [go]
script := []scenario.ScriptStep{
	scenario.User("I need help"),
	scenario.Agent(),

	// Checkpoint: if criteria pass, continue; if any fail, stop immediately
	scenario.Judge(scenario.WithCriteria("Agent should ask clarifying questions")),

	scenario.User(),
	scenario.Agent(),

	// Final evaluation with remaining criteria
	scenario.Judge(scenario.WithCriteria(
		"Agent provides a helpful response",
		"Agent does not make things up",
	)),
}
```

:::

Each `scenario.judge()` with inline criteria acts as a **checkpoint**: if all criteria pass, the script continues; if any fail, the scenario stops immediately with a failure. When the script ends after all checkpoints pass, the scenario succeeds with all accumulated passed criteria.

### <RefLink link={{ python: "script.html#scenario.script.proceed", typescript: "functions/proceed.html" }}>`scenario.proceed()`</RefLink>

Let the conversation flow automatically for a specified number of turns:

```python
script=[
    scenario.user("Start the conversation"),
    scenario.agent(),

    # Let it proceed for 3 turns
    scenario.proceed(turns=3),

    # Then take control again
    scenario.user("Final question"),
    scenario.agent(),
    scenario.succeed()
]
```

### <RefLink link={{ python: "script.html#scenario.script.succeed", typescript: "functions/succeed.html" }}>`scenario.succeed()`</RefLink> and <RefLink link={{ python: "script.html#scenario.script.fail", typescript: "functions/fail.html" }}>`scenario.fail()`</RefLink>

## Custom Steps and Evaluations

### Custom Assertion Functions

Add custom logic at any point in the conversation:

:::code-group

```python [python]
def check_tool_usage(state: scenario.ScenarioState) -> None:
    """Verify agent called the required tool"""
    assert state.has_tool_call("get_weather"), "Agent should have called weather tool"

    # Get the arguments of the last call
    args = state.tool_calls("get_weather").last.input
    assert args and "location" in args, "Weather tool should include location"

def verify_response_quality(state: scenario.ScenarioState) -> None:
    """Check response meets quality standards"""
    last_message = state.last_message()
    content = last_message.get("content", "")

    assert len(content) > 10, "Response should be substantial"
    assert "sorry" not in content.lower(), "Agent shouldn't apologize unnecessarily"

script=[
    scenario.user("What's the weather in Paris?"),
    scenario.agent(),
    check_tool_usage,        # Custom assertion
    verify_response_quality, # Another custom check
    scenario.succeed()
]
```

```typescript [typescript]

async function check_tool_usage(state: scenario.ScenarioExecutionStateLike) {
    // Verify agent called the required tool
    if (!state.hasToolCall("get_weather")) throw new Error("Agent should have called weather tool");

    // Get the arguments of the last call
    const args = state.toolCalls("get_weather").last?.input as { location?: string } | undefined;
    if (!args?.location) throw new Error("Weather tool should include location");
}

async function verify_response_quality(state: scenario.ScenarioExecutionStateLike) {

    const lastMessage = state.lastMessage();
    const content = lastMessage.content;

    if (content.length <= 10) throw new Error("Response should be substantial");
    if (content.toLowerCase().includes("sorry")) throw new Error("Agent shouldn't apologize unnecessarily");

const script = [
  scenario.user("What's the weather in Paris?"),
  scenario.agent(),
  check_tool_usage,
  verify_response_quality,
  scenario.succeed(),
];
```

```go [go]
func checkToolUsage(ctx context.Context, ex *scenario.Executor) error {
	state := ex.State()
	// Verify agent called the required tool
	if !state.HasToolCall("get_weather") {
		return &scenario.AssertionError{Message: "Agent should have called weather tool"}
	}

	// Get the arguments of the last call
	last, _ := state.ToolCalls("get_weather").Last()
	args, _ := last.Input.(map[string]any)
	if location, _ := args["location"].(string); location == "" {
		return &scenario.AssertionError{Message: "Weather tool should include location"}
	}
	return nil
}

func verifyResponseQuality(ctx context.Context, ex *scenario.Executor) error {
	lastMessage, err := ex.State().LastMessage()
	if err != nil {
		return err
	}
	content := lastMessage.Text()

	if len(content) <= 10 {
		return &scenario.AssertionError{Message: "Response should be substantial"}
	}
	if strings.Contains(strings.ToLower(content), "sorry") {
		return &scenario.AssertionError{Message: "Agent shouldn't apologize unnecessarily"}
	}
	return nil
}

script := []scenario.ScriptStep{
	scenario.User("What's the weather in Paris?"),
	scenario.Agent(),
	scenario.StepFunc(checkToolUsage),
	scenario.StepFunc(verifyResponseQuality),
	scenario.Succeed(),
}
```

:::

In Go, a custom step is a `scenario.StepFunc`. Return a `*scenario.AssertionError` (or `scenario.Assert(cond, msg)`) to fail the scenario with that message. Any other error ends the run as an error.

### Reading the Run State

A custom step receives the scenario state. Besides the messages, the state exposes the scenario definition and the trace of the run so far, the same accessors an [evaluator mapping](/advanced/evaluators) reads:

| Python | TypeScript | Go | Reads |
| --- | --- | --- | --- |
| `state.messages` | `state.messages` | `state.Messages()` | The conversation so far |
| `state.last_message()`, `state.last_user_message()` | `state.lastMessage()`, `state.lastUserMessage()`, `state.lastAgentMessage()` | `state.LastMessage()`, `state.LastUserMessage()`, `state.LastAgentMessage()` | One message |
| `state.first_user_message()`, `state.last_agent_message()` | `state.firstUserMessage()` | `state.FirstUserMessage()` | The text of one message |
| `state.transcript()` | `state.transcript()` | `state.Transcript()` | One `role: content` line per message |
| `state.has_tool_call(name)` | `state.hasToolCall(name)` | `state.HasToolCall(name)` | Whether the tool was called |
| `state.tool_calls(name)` | `state.toolCalls(name)` | `state.ToolCalls(name)` | Every call of the tool so far, from the messages and the spans, in start order |
| `state.fields`, `state.field(name)` | `state.fields`, `state.field(name)` | `state.Fields()`, `state.Field(name)` | The values passed as `fields` to `run` |
| `state.criteria` | `state.criteria` | `state.Criteria()` | The judge criteria |
| `state.contexts` | `state.contexts` | `state.Contexts()` | Every chunk the agent retrieved, from the rag spans |
| `state.spans` | `state.spans` | `state.Spans()` | Every span of every trace of the run collected so far, in start order |
| `state.traces` | `state.traces` | `state.Traces()` | One entry per trace id on the messages: `id`, `spans`, `tool_calls(name)` |
| `state.turns` | `state.turns` | `state.Turns()` | One entry per turn: `index`, `messages`, `trace`, `tool_calls(name)` |

In Go, a custom step reads the state with `ex.State()`, and the message getters return the message and an error when there is none.

`tool_calls(name)` is a collection: `first` and `last` pick one call, `inputs` and `outputs` list the arguments and results of every call, and it iterates, indexes and counts. A call has `name`, `input`, `output`, `turn` and `source` (`message` or `span`). In Python, `first` and `last` of a tool that was never called are an empty, falsy call whose `input` and `output` are `None`; in TypeScript they are `undefined`; in Go, `First()` and `Last()` return `false` as their second value. `last_tool_call(name)` keeps working and reads the messages only.

The whole run is the default scope; `turns[i]` and `traces[i]` narrow it. `state.spans` reads what the span collector holds at that moment and never fetches remote traces.

:::code-group

```python [python]
def check_sql(state: scenario.ScenarioState) -> None:
    calls = state.tool_calls("run_sql")
    assert len(calls) == 1
    assert calls.last.input["sql"].startswith("SELECT")
    assert state.field("golden_sql") is not None
    assert state.turns[-1].tool_calls("run_sql").first.output is not None
```

```typescript [typescript]
function checkSql(state: scenario.ScenarioExecutionStateLike) {
  const calls = state.toolCalls("run_sql");
  if (calls.length !== 1) throw new Error("Expected one run_sql call");
  const sql = (calls.last?.input as { sql: string }).sql;
  if (!sql.startsWith("SELECT")) throw new Error("Expected a SELECT query");
  if (state.field("golden_sql") === undefined) throw new Error("Expected the golden_sql field");
  const lastTurn = state.turns.at(-1);
  if (lastTurn?.toolCalls("run_sql").first?.output === undefined) throw new Error("Expected a run_sql result");
}
```

```go [go]
func checkSQL(ctx context.Context, ex *scenario.Executor) error {
	state := ex.State()
	calls := state.ToolCalls("run_sql")
	if len(calls) != 1 {
		return &scenario.AssertionError{Message: "Expected one run_sql call"}
	}
	last, _ := calls.Last()
	input, _ := last.Input.(map[string]any)
	if sql, _ := input["sql"].(string); !strings.HasPrefix(sql, "SELECT") {
		return &scenario.AssertionError{Message: "Expected a SELECT query"}
	}
	if _, ok := state.Field("golden_sql"); !ok {
		return &scenario.AssertionError{Message: "Expected the golden_sql field"}
	}
	turns := state.Turns()
	if len(turns) == 0 {
		return &scenario.AssertionError{Message: "Expected a run_sql result"}
	}
	if first, ok := turns[len(turns)-1].ToolCalls("run_sql").First(); !ok || first.Output == nil {
		return &scenario.AssertionError{Message: "Expected a run_sql result"}
	}
	return nil
}
```

### Async Custom Steps

Custom steps can be async for external API calls:

```python [python]
async def external_evaluation(state: scenario.ScenarioState) -> None:
    """Run external evaluation service"""
    last_response = state.last_message().get("content", "")

    # Call external service
    eval_result = await external_evaluator.evaluate(
```

```typescript [typescript]
async function externalEval(state: scenario.ScenarioState) {

    const lastResponse = state.lastMessage().content;

    // Call external service
```

```go [go]
// Go steps are plain blocking functions: call the service directly and pass ctx along.
func externalEval(ctx context.Context, ex *scenario.Executor) error {
	state := ex.State()
	// Run external evaluation service
	lastResponse, err := state.LastMessage()
	if err != nil {
		return err
	}
	lastUserMessage, err := state.LastUserMessage()
	if err != nil {
		return err
	}

	// Call external service
	evalResult, err := externalEvaluator.Evaluate(ctx, lastUserMessage.Text(), lastResponse.Text())
	if err != nil {
		return err
	}

	if evalResult.Score <= 0.8 {
		return &scenario.AssertionError{Message: fmt.Sprintf("Quality score too low: %v", evalResult.Score)}
	}
	return nil
}

script := []scenario.ScriptStep{
	scenario.User("Complex query"),
	scenario.Agent(),
	scenario.StepFunc(externalEval),
	scenario.Proceed(),
}
```

### Conditional Logic

Use conditional logic to branch based on conversation state:

:::code-group

```python [python]
def conditional_check(state: scenario.ScenarioState) -> Optional[scenario.ScenarioResult]:
    """Branch logic based on agent response"""
    last_message = state.last_message()
    content = last_message.get("content", "").lower()

    if "i don't know" in content:
        return scenario.ScenarioResult(
            success=False,
            messages=state.messages,
            reasoning="Agent should not give up so easily"
        )
    elif "let me help" in content:
        # Continue the conversation
        return None
    else:
        # Force a specific follow-up
        state.add_message({
            "role": "user",
            "content": "Can you be more specific?"
        })
        return None

script=[
    scenario.user("I have a problem"),
    scenario.agent(),
    conditional_check,  # Returns ScenarioResult or None
    scenario.proceed()
]
```

```typescript [typescript]
async function conditionalCheck(state: scenario.ScenarioState) {
    // Branch logic based on agent response
    const lastMessage = state.lastMessage();
    const content = lastMessage.content.toLowerCase();

    if (content.includes("i don't know")) {
        return scenario.ScenarioResult(
            success: false,
            messages: state.messages,
            reasoning: "Agent should not give up so easily"
        );
    } else if (content.includes("let me help")) {
        // Continue the conversation
        return null;
    } else {
        // Force a specific follow-up
        state.addMessage({
            role: "user",
```
